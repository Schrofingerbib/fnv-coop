cmake_minimum_required(VERSION 3.20)
project(fnvcoop_fuzz LANGUAGES CXX)
# Standalone on purpose: it does not touch synccore/CMakeLists.txt, so it cannot break the L4 build.
# Build: synccore\fuzz\run_tests.bat   (Win32 MSVC + Ninja, build dir build\T2b-fuzz)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(SC ${CMAKE_CURRENT_SOURCE_DIR}/..)

set(FUZZ_SRC scenario.cpp oracles.cpp runner.cpp mini_target.cpp targets.cpp ${SC}/src/transport.cpp)
set(FUZZ_DEFS "")
set(FUZZ_LIBS "")
# Flight-format traces (T1a) when the recorder is present.
option(FUZZ_FLIGHT "write .nvfr traces with synccore/flight (T1a)" ON)
if(FUZZ_FLIGHT AND EXISTS ${SC}/flight/flight.cpp)
  set(FNVCOOP_CORE_SRC wire shadow policy storylog protocol rules baseline replicator transport transport_util session ownership)
  set(CORE "")
  foreach(f ${FNVCOOP_CORE_SRC})
    list(APPEND CORE ${SC}/src/${f}.cpp)
  endforeach()
  list(REMOVE_ITEM FUZZ_SRC ${SC}/src/transport.cpp)
  list(APPEND FUZZ_SRC ${CORE} ${SC}/flight/flight.cpp)
  list(APPEND FUZZ_DEFS FNVCOOP_FUZZ_FLIGHT)
endif()
# T2a's engine model adapter (the "em" target): ON by default now that the model fuzzes clean; -DFUZZ_EM=OFF (FUZZ_EM=0 for the .bat files) builds the mini target only.
option(FUZZ_EM "build the em target (synccore/enginemodel)" ON)
if(FUZZ_EM AND EXISTS ${SC}/enginemodel/src/cluster.cpp)
  file(GLOB EM_SRC ${SC}/enginemodel/src/*.cpp)
  if(NOT FUZZ_FLIGHT)
    foreach(f wire shadow policy storylog protocol rules baseline replicator transport_util session ownership)
      list(APPEND EM_SRC ${SC}/src/${f}.cpp)
    endforeach()
  endif()
  list(APPEND FUZZ_SRC target_em.cpp ${EM_SRC} ${SC}/combat/combat.cpp)   # combat-core: the rules the engine model runs
  list(APPEND FUZZ_DEFS FNVCOOP_FUZZ_EM)
endif()

add_library(simfuzz_lib STATIC ${FUZZ_SRC})
target_include_directories(simfuzz_lib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${SC}/include ${SC} ${SC}/enginemodel/include)
target_compile_definitions(simfuzz_lib PUBLIC ${FUZZ_DEFS})

add_executable(fnvcoop_simfuzz simfuzz_main.cpp)
target_link_libraries(fnvcoop_simfuzz PRIVATE simfuzz_lib)
add_executable(fnvcoop_fuzz_tests tests/fuzz_tests.cpp ${SC}/tests/main.cpp)
target_include_directories(fnvcoop_fuzz_tests PRIVATE ${SC}/tests)
target_compile_definitions(fnvcoop_fuzz_tests PRIVATE FUZZ_REGRESS_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../../scenarios/simfuzz")
target_link_libraries(fnvcoop_fuzz_tests PRIVATE simfuzz_lib)

if(MSVC)
  foreach(t simfuzz_lib fnvcoop_simfuzz fnvcoop_fuzz_tests)
    target_compile_options(${t} PRIVATE /W4 /permissive- /utf-8 /wd4456 /D_CRT_SECURE_NO_WARNINGS)
  endforeach()
endif()

enable_testing()
add_test(NAME fuzz_unit COMMAND fnvcoop_fuzz_tests --group fuzz)
