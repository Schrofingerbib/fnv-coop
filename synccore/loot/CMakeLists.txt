# synccore/loot: loot, containers, dropped items and physics clutter (FULL-SYNC-PLAN 3.5), game-free. Included from synccore/CMakeLists.txt.
#   fnvcoop_loot     production library (twins compiled out): what plugin/loot links
#   fnvcoop_loot_rt  the same sources with FNVCOOP_RUNTIME_MUT: the test exe picks a mutation twin with FNVCOOP_MUT=<n> at run time
#   loot_tests       unit tests + model tests + fuzz; twins 101..137 and 170..173 must each fail it (synccore\loot\run_tests.cmd runs them all)
set(LOOT_MUT_INC ${CMAKE_CURRENT_LIST_DIR}/../../tools/mut)
foreach(sfx "" "_rt")
  add_library(fnvcoop_loot${sfx} STATIC ${CMAKE_CURRENT_LIST_DIR}/loot.cpp ${CMAKE_CURRENT_LIST_DIR}/loot_wire.cpp)
  target_include_directories(fnvcoop_loot${sfx} PUBLIC ${CMAKE_CURRENT_LIST_DIR}/.. ${LOOT_MUT_INC})
  target_link_libraries(fnvcoop_loot${sfx} PUBLIC fnvcoop)
  if(sfx STREQUAL "_rt")
    target_compile_definitions(fnvcoop_loot${sfx} PUBLIC FNVCOOP_RUNTIME_MUT)
  endif()
  if(MSVC)
    target_compile_options(fnvcoop_loot${sfx} PRIVATE /W4 /permissive- /utf-8 /wd4456 /D_CRT_SECURE_NO_WARNINGS)
  else()
    target_compile_options(fnvcoop_loot${sfx} PRIVATE -Wall -Wextra)
  endif()
endforeach()

add_executable(loot_tests tests/test_host.cpp tests/test_client.cpp tests/test_wire.cpp tests/test_model.cpp ${CMAKE_CURRENT_LIST_DIR}/../tests/main.cpp)
target_include_directories(loot_tests PRIVATE ${CMAKE_CURRENT_LIST_DIR}/../tests ${CMAKE_CURRENT_LIST_DIR}/tests)
target_link_libraries(loot_tests PRIVATE fnvcoop_loot_rt)
if(MSVC)
  target_compile_options(loot_tests PRIVATE /W4 /permissive- /utf-8 /wd4456 /D_CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(loot_tests PRIVATE -Wall -Wextra)
endif()
if(NOT CMAKE_RUNTIME_OUTPUT_DIRECTORY)
  foreach(cfg "" _DEBUG _RELEASE _RELWITHDEBINFO _MINSIZEREL)
    set_target_properties(loot_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY${cfg} "${CMAKE_BINARY_DIR}/loot_tests")
  endforeach()
endif()

add_test(NAME loot_core_unit COMMAND loot_tests)
foreach(v RANGE 101 137)
  add_test(NAME loot_core_mut${v} COMMAND loot_tests)
  set_tests_properties(loot_core_mut${v} PROPERTIES WILL_FAIL TRUE ENVIRONMENT "FNVCOOP_MUT=${v}")
endforeach()
foreach(v 170 171 172 173)
  add_test(NAME loot_core_mut${v} COMMAND loot_tests)
  set_tests_properties(loot_core_mut${v} PROPERTIES WILL_FAIL TRUE ENVIRONMENT "FNVCOOP_MUT=${v}")
endforeach()
