# T5 chaos mirror: header-only impairment model + cross-check driver against tools/chaos/impair.py.
add_executable(chaos_xcheck xcheck.cpp)
target_include_directories(chaos_xcheck PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
# Twin builds: each CHAOS_MUTANT value breaks one behaviour in the C++ mirror; the cross-check must then fail.
foreach(m 1 2 3 4)
  add_executable(chaos_xcheck_m${m} xcheck.cpp)
  target_include_directories(chaos_xcheck_m${m} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_definitions(chaos_xcheck_m${m} PRIVATE CHAOS_MUTANT=${m})
endforeach()
if(MSVC)
  foreach(t chaos_xcheck chaos_xcheck_m1 chaos_xcheck_m2 chaos_xcheck_m3 chaos_xcheck_m4)
    target_compile_options(${t} PRIVATE /W4 /permissive- /utf-8 /fp:precise)
  endforeach()
endif()
if(FNVCOOP_PYTHON)
  set(XC ${FNVCOOP_PYTHON} -I ${CMAKE_CURRENT_SOURCE_DIR}/../../tools/chaos/xcheck.py)
  add_test(NAME chaos_header_fresh COMMAND ${FNVCOOP_PYTHON} -I ${CMAKE_CURRENT_SOURCE_DIR}/../../tools/chaos/gen_header.py --check)
  add_test(NAME twin_chaos_header_stale COMMAND ${FNVCOOP_PYTHON} -I ${CMAKE_CURRENT_SOURCE_DIR}/../../tools/chaos/gen_header.py --check --mutate)
  set_tests_properties(twin_chaos_header_stale PROPERTIES WILL_FAIL TRUE)
  add_test(NAME chaos_xcheck COMMAND ${XC} $<TARGET_FILE:chaos_xcheck>)
  foreach(m 1 2 3 4)
    add_test(NAME twin_chaos_xcheck_m${m} COMMAND ${XC} $<TARGET_FILE:chaos_xcheck_m${m}>)
    set_tests_properties(twin_chaos_xcheck_m${m} PROPERTIES WILL_FAIL TRUE)
  endforeach()
endif()
