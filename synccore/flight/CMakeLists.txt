# T1a flight recorder: game-free library + tests (+ mutation twins). Included from synccore/CMakeLists.txt.
add_library(fnvcoop_flight STATIC flight.cpp)
target_include_directories(fnvcoop_flight PUBLIC ${CMAKE_CURRENT_LIST_DIR}/..)
target_link_libraries(fnvcoop_flight PUBLIC fnvcoop)

add_executable(flight_tests tests/test_flight.cpp ${CMAKE_CURRENT_LIST_DIR}/../tests/main.cpp)
target_link_libraries(flight_tests PRIVATE fnvcoop_flight fnvcoop_sim)
if(MSVC)
  foreach(t fnvcoop_flight flight_tests)
    target_compile_options(${t} PRIVATE /W4 /permissive- /utf-8 /wd4456 /D_CRT_SECURE_NO_WARNINGS)
  endforeach()
else()
  target_compile_options(fnvcoop_flight PRIVATE -Wall -Wextra)
  target_compile_options(flight_tests PRIVATE -Wall -Wextra)
endif()

set(FLIGHT_TEST_ENV "FNVCOOP_FLIGHT_GOLDEN=${CMAKE_CURRENT_LIST_DIR}/golden")
add_test(NAME flight_unit COMMAND flight_tests --group flight)
set_tests_properties(flight_unit PROPERTIES ENVIRONMENT "${FLIGHT_TEST_ENV}")
add_test(NAME flight_perf COMMAND flight_tests --group perf)
foreach(twin drop_newest no_net_in zero_seq no_trunc_flag no_dropped_count no_applied_flags no_derived_digest buckets_ignore_value detail_drops_key incremental_misses_dirty)
  add_test(NAME twin_flight_${twin} COMMAND flight_tests --group twin --name ${twin})
endforeach()
