# T3a hot-reload loader. Works two ways:
#   * included by plugin/CMakeLists.txt when -DNVCOOP_SPLIT_LOADER=ON: adds nvloader_host and the
#     `nvcoop_loader` target (the permanent nvcoop.dll; the old `nvcoop` target becomes nvcoop_mod.dll).
#     Game-free tests are added too when -DNVCOOP_LOADER_TESTS=ON.
#   * standalone, game-free tests only (x86 like the game; mutation twins included):
#       plugin\loader\run_tests.cmd        (prints ALL OK)
cmake_minimum_required(VERSION 3.20)
set(LOADER_DIR "${CMAKE_CURRENT_LIST_DIR}")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_LIST_DIR)
  project(nvcoop_loader_tests CXX)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
  set(LOADER_STANDALONE ON)
  set(LOADER_OUT "${LOADER_DIR}/../../out/T3a" CACHE PATH "Where the loader test binaries land")
  foreach(cfg "" _DEBUG _RELEASE _RELWITHDEBINFO _MINSIZEREL)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY${cfg} "${LOADER_OUT}")
    set(CMAKE_LIBRARY_OUTPUT_DIRECTORY${cfg} "${LOADER_OUT}")
  endforeach()
  # the registry (events, methods) comes from the L1 sources when we are not inside the plugin build
  add_library(nvloader_api STATIC ${LOADER_DIR}/../common/nvcoop_api.cpp)
  target_include_directories(nvloader_api PUBLIC ${LOADER_DIR}/.. ${LOADER_DIR}/../common)
  target_compile_definitions(nvloader_api PUBLIC _CRT_SECURE_NO_WARNINGS)
  set(LOADER_API_LIB nvloader_api)
else()
  set(LOADER_STANDALONE OFF)
  set(LOADER_API_LIB nvcoop_core)
endif()

function(loader_flags tgt)
  if(MSVC)
    target_compile_options(${tgt} PRIVATE /W4 /EHsc /permissive- /utf-8 /wd4100)
  endif()
  target_compile_definitions(${tgt} PRIVATE _CRT_SECURE_NO_WARNINGS WIN32_LEAN_AND_MEAN)
endfunction()

# nvloader_host = CoreHost; the _m* twins each break one safety rule (see loader_host.cpp)
set(LOADER_MUTANTS NOCOPY NOPURGE NOVETO NOROLLBACK)
function(loader_host_lib name)
  add_library(${name} STATIC ${LOADER_DIR}/loader_host.cpp)
  target_include_directories(${name} PUBLIC ${LOADER_DIR} ${LOADER_DIR}/..)
  target_link_libraries(${name} PUBLIC ${LOADER_API_LIB})
  loader_flags(${name})
  foreach(d ${ARGN})
    target_compile_definitions(${name} PRIVATE ${d})
  endforeach()
endfunction()
loader_host_lib(nvloader_host)

# ---- the permanent loader plugin (nvcoop.dll): main.cpp built with NVCOOP_LOADER ----
if(NOT LOADER_STANDALONE)
  add_library(nvcoop_loader SHARED ${LOADER_DIR}/../main.cpp)
  set_target_properties(nvcoop_loader PROPERTIES PREFIX "" OUTPUT_NAME "nvcoop")
  target_link_libraries(nvcoop_loader PRIVATE nvcoop_core nvloader_host)
  if(TARGET nvsplit_focus)  # plugin/splitscreen focus guard: window subclass, must live in this permanent DLL
    target_link_libraries(nvcoop_loader PRIVATE nvsplit_focus)
    target_compile_definitions(nvcoop_loader PRIVATE NVCOOP_WITH_FOCUS_GUARD=1)
  endif()
  target_include_directories(nvcoop_loader PUBLIC "${LOADER_DIR}/.." "${FNVCOOP_NVSE_DIR}" "${FNVCOOP_NVSE_DIR}/nvse" "${FNVCOOP_NVSE_DIR}/nvse/nvse")
  target_compile_definitions(nvcoop_loader PRIVATE NVCOOP_LOADER=1 RUNTIME=1 RUNTIME_VERSION=0x040020D0 WIN32 _WINDOWS _CRT_SECURE_NO_WARNINGS)
  if(MSVC)
    target_compile_options(nvcoop_loader PRIVATE "SHELL:/FI ${FNVCOOP_NVSE_DIR}/nvse/nvse/prefix.h" /W3 /EHsc /permissive- /Zc:__cplusplus)
  endif()
endif()

# ---- game-free tests: stub cores + test exe (+ mutation twins) ----
option(NVCOOP_LOADER_TESTS "Build the loader's game-free tests and stub cores" OFF)
if(LOADER_STANDALONE OR NVCOOP_LOADER_TESTS)
  function(stub_core name)
    add_library(${name} SHARED ${LOADER_DIR}/tests/stub_core.cpp)
    set_target_properties(${name} PROPERTIES PREFIX "" OUTPUT_NAME "${name}")
    target_include_directories(${name} PRIVATE ${LOADER_DIR})
    loader_flags(${name})
    foreach(d ${ARGN})
      target_compile_definitions(${name} PRIVATE ${d})
    endforeach()
  endfunction()
  stub_core(nvstub_v1 STUB_VERSION=1)
  stub_core(nvstub_v2 STUB_VERSION=2)
  stub_core(nvstub_failinit STUB_VERSION=3 STUB_FAIL_INIT)
  stub_core(nvstub_noexport STUB_VERSION=4 STUB_NO_EXPORT)
  stub_core(nvstub_badabi STUB_VERSION=5 STUB_BAD_ABI)

  function(loader_test name hostlib)
    add_executable(${name} ${LOADER_DIR}/tests/test_loader.cpp)
    target_link_libraries(${name} PRIVATE ${hostlib})
    loader_flags(${name})
    add_dependencies(${name} nvstub_v1 nvstub_v2 nvstub_failinit nvstub_noexport nvstub_badabi)
  endfunction()
  loader_test(nvloader_test nvloader_host)
  foreach(m ${LOADER_MUTANTS})
    string(TOLOWER ${m} lm)
    loader_host_lib(nvloader_host_${lm} NVLOADER_MUTANT_${m})
    loader_test(nvloader_test_${lm} nvloader_host_${lm})
  endforeach()
endif()
